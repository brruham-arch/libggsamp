cmake_minimum_required(VERSION 3.18)
project(ggsamp C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LUA ${CMAKE_SOURCE_DIR}/lua)
set(LS lapi lcode lctype ldebug ldo ldump lfunc lgc llex lmem lobject lopcodes lparser lstate lstring ltable ltm lundump lvm lzio lauxlib lbaselib lcorolib ldblib liolib lmathlib loadlib loslib lstrlib ltablib lutf8lib linit)
set(SRCS src/main.cpp)
foreach(f ${LS})
  list(APPEND SRCS ${LUA}/${f}.c)
endforeach()
add_library(ggsamp SHARED ${SRCS})
target_include_directories(ggsamp PRIVATE ${LUA})
target_compile_options(ggsamp PRIVATE -O2 -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_libraries(ggsamp log dl)
target_link_options(ggsamp PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic -Wl,--gc-sections)
